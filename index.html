        function shuffleArray(array) {

            const arr = [...array];

            for (let i = arr.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [arr[i], arr[j]] = [arr[j], arr[i]];

            }

            return arr;

        }

        function updateProgress() {

            let totalItems = 0, currentProgress = 0, titleText = "STAGE PROGRESS";

            if (currentStage === 0) {

                totalItems = activeStage0Data.length;

                currentProgress = s0Idx;

                titleText = "STAGE 0: WIRE CUTTING";

            } else if (currentStage === 1) {

                totalItems = activeStage1Data.length;

                currentProgress = s1Idx;

                titleText = "STAGE 1: CASH BOOK LEVER";

            } else if (currentStage === 2) {

                totalItems = activeStage2Data.length;

                currentProgress = s2Idx;

                titleText = "STAGE 2: ERROR CORRECTION";

            } else if (currentStage === 3) {

                totalItems = activeStage3Data.length;

                currentProgress = s3Idx;

                titleText = "STAGE 3: BRS CIRCUITS";

            } else if (currentStage === 4) {

                totalItems = activeStage4Data.length;

                currentProgress = s4Idx;

                titleText = "STAGE 4: UNADJUSTED BRS";

            } else {

                titleText = "START PAGE";

                totalItems = 1;

                currentProgress = 0;

            }

 

            document.getElementById('stageTitleProgress').textContent = titleText;

            document.getElementById('progressText').textContent = `${currentProgress} / ${totalItems}`;

            const pct = Math.round((currentProgress / Math.max(totalItems, 1)) * 100);

            document.getElementById('progressBar').style.width = `${pct}%`;

        }

 

        function startTimer() {

            if (timerInterval) clearInterval(timerInterval);

            timerInterval = setInterval(() => {

                if (totalTimeSeconds <= 0) {

                    clearInterval(timerInterval);

                    document.getElementById('timerDisplay').textContent = "00:00.00";

                    playSound('error');

                    alert("💥 BOOM! The detonation timer reached 0. Click Reset to retry.");

                    return;

                }

                totalTimeSeconds -= 0.1;

                const m = Math.floor(totalTimeSeconds / 60);

                const s = Math.floor(totalTimeSeconds % 60);

                const ms = Math.floor((totalTimeSeconds % 1) * 100);

                document.getElementById('timerDisplay').textContent =

                    `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}.${String(ms).padStart(2, '0')}`;

            }, 100);

        }
